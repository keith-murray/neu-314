\section*{Problem 1: Linear functions}
\begin{enumerate}
    \item[(a)] Is $\cos x$ a linear function?
    \item[(b)] Are polynomials of the form $f(x)=ax^2+bx+c$ linear functions?
\end{enumerate}

\section*{Problem 2: Dimension of a Span}
Find the dimension of the span of each set of vectors in $\mathbb{R}^4$.
\begin{enumerate}
    \item[(a)] $\vec{v}_1=\begin{bmatrix}1\\2\\0\\1\end{bmatrix},\;
    \vec{v}_2=\begin{bmatrix}2\\4\\1\\0\end{bmatrix},\;
    \vec{v}_3=\begin{bmatrix}3\\6\\1\\1\end{bmatrix},\;
    \vec{v}_4=\begin{bmatrix}0\\0\\1\\-2\end{bmatrix}$
    \vspace{0.5em}
    \item[(b)] $\vec{w}_1=\begin{bmatrix}1\\1\\0\\0\end{bmatrix},\;
    \vec{w}_2=\begin{bmatrix}0\\1\\1\\0\end{bmatrix},\;
    \vec{w}_3=\begin{bmatrix}0\\0\\1\\1\end{bmatrix},\;
    \vec{w}_4=\begin{bmatrix}1\\0\\0\\1\end{bmatrix}$
\end{enumerate}

\section*{Problem 3: Rank}
\begin{enumerate}
    \item[(a)] Find the rank of
    \[
    A=\begin{bmatrix}
        1 & 2 & 1 & 3\\
        2 & 4 & 3 & 7\\
        1 & 2 & 2 & 4
    \end{bmatrix}
    \]
    \item[(b)] Let
    \[
    C=\begin{bmatrix}
        1 & 1 & 1\\
        1 & 2 & 3\\
        2 & 3 & c
    \end{bmatrix}
    \]
    For which values of $c$ does $C$ have rank $3$, and for which does it have rank $2$?
\end{enumerate}

\section*{Problem 4: Null Space}
\begin{enumerate}
    \item[(a)] Let $A$ be a $5\times 7$ matrix. What is the largest possible rank of $A$? What is the smallest dimension of the null space of $A$?
    \item[(b)] True or false (give a short reason): if a $4\times 4$ matrix has linearly independent columns, then its null space is $\{\vec{0}\}$ and its rows are also linearly independent.
\end{enumerate}
